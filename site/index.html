A few lines, in Rust or JavaScript.

use taktus::{beats_of, tempo_of, Ear};

// Mono samples in, onsets out: a hundred values a second.
let mut ear = Ear::new(44_100);
let onsets: Vec<f32> = samples.iter().filter_map(|s| ear.hear(*s)).collect();

if let Some(beat) = tempo_of(&onsets) {
    let beats = beats_of(&onsets, beat.period); // seconds from the start
    println!("{:.1} beats a minute, {} beats", beat.bpm(), beats.len());
}
import { Ear, beatsOf, tempoOf } from "taktus";

// Mono samples in, onsets out: a hundred values a second.
const ear = new Ear(audio.sampleRate);
const onsets = ear.hear(audio.getChannelData(0));

const beat = tempoOf(onsets);
if (beat) {
  const beats = beatsOf(onsets, beat.period); // seconds from the start
  console.log(`${beat.bpm.toFixed(1)} beats a minute, ${beats.length} beats`);
}
